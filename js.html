Что такое JavaScript?

JavaScript (JS) — это язык программирования, который работает в браузере. Он позволяет:

  • Изменять содержимое страницы
  • Менять стили элементов
  • Обрабатывать действия пользователя (клики, ввод)
  • Отправлять запросы на сервер
  • Работать с данными
Важно: JavaScript — это язык программирования. В отличие от HTML и CSS, в нём есть логика, условия, циклы и функции.

Переменные

let

Переменная, которую можно изменять

let name = 'Анна';
name = 'Мария';
console.log(name);
const

Константа — значение нельзя изменить

const PI = 3.14;
// PI = 3.15 — ошибка!
var

Устаревший способ (не использовать)

// Устарело!
// Используй let или const

Правило: Используй const по умолчанию, let только если значение будет меняться. var не используй!

Типы данных

Строки (string)

Текст в кавычках

let text = 'Привет!';
let name = "Анна";
Числа (number)

Целые и дробные числа

let age = 25;
let price = 99.99;
Булевы (boolean)

true или false

let isReady = true;
let isLogged = false;
null и undefined

Пустое значение или не определено

let user = null;
let something; // undefined

Определение типа:

typeof 'Привет'; // 'string'
typeof 42; // 'number'
typeof true; // 'boolean'
typeof null; // 'object' (ошибка JS)

Условные операторы (if / else)

if

Выполняется, если условие true

let age = 18;
if (age >= 18) {
  console.log('Доступ разрешён');
}
if / else

Одно из двух

let age = 16;
if (age >= 18) {
  console.log('Доступ разрешён');
} else {
  console.log('Доступ запрещён');
}
if / else if / else

Несколько условий

let num = 0;
if (num > 0) {
  console.log('Положительное');
} else if (num < 0) {
  console.log('Отрицательное');
} else {
  console.log('Ноль');
}
Логические операторы

&& (и), || (или), ! (не)

if (age >= 18 && hasPass) {
  console.log('Пропуск');
}

if (isAdmin || isOwner) {
  console.log('Доступ');
}

Циклы

for

Цикл с известным количеством повторений

for (let i = 1; i <= 5; i++) {
  console.log(i); // 1, 2, 3, 4, 5
}
while

Цикл с условием

let i = 1;
while (i <= 5) {
  console.log(i);
  i++;
}
for / of (для массивов)

Перебор элементов массива

let arr = [1, 2, 3];
for (let item of arr) {
  console.log(item);
}
for / in (для объектов)

Перебор ключей объекта

let obj = {a: 1, b: 2};
for (let key in obj) {
  console.log(key, obj[key]);
}

Массивы

Массив — это упорядоченный набор данных. Индексация начинается с 0.

let arr = [1, 2, 3, 4, 5];
console.log(arr[0]); // 1
console.log(arr[2]); // 3
console.log(arr.length); // 5
Методы массивов
arr.push(6); // добавить в конец
arr.pop(); // удалить последний
arr.unshift(0); // добавить в начало
arr.shift(); // удалить первый
Методы преобразования
arr.map(x => x * 2); // умножить каждый
arr.filter(x => x > 2); // оставить > 2
arr.reduce((sum, x) => sum + x); // сумма
arr.sort((a, b) => a - b); // сортировка

Объекты

Объект — это набор пар «ключ: значение».

let user = {
  name: 'Анна',
  age: 25,
  city: 'Москва'
};

console.log(user.name); // 'Анна'
console.log(user['age']); // 25
Работа с объектами
// Добавление свойства
user.email = 'anna@mail.ru';

// Удаление свойства
delete user.city;

// Получение ключей
Object.keys(user); // ['name', 'age']

// Получение значений
Object.values(user); // ['Анна', 25]
Объекты с вложенностью
let user = {
  name: 'Анна',
  address: {
    city: 'Москва',
    street: 'Тверская'
  }
};

console.log(user.address.city);

Функции

Объявление функции
function greet(name) {
  return 'Привет, ' + name;
}

greet('Анна'); // 'Привет, Анна'
Стрелочные функции
const greet = (name) => {
  return 'Привет, ' + name;
};

// Короткая запись
const double = x => x * 2;
Функции с параметрами
function sum(a, b) {
  return a + b;
}

// Параметры по умолчанию
function greet(name = 'Гость') {
  return 'Привет, ' + name;
}

Работа с датами

// Текущая дата и время
let now = new Date();

// Создание даты
let date = new Date(2026, 0, 30); // 30 января 2026

// Методы даты
date.getFullYear(); // 2026
date.getMonth(); // 0 (январь)
date.getDate(); // 30
date.getDay(); // день недели (0 - воскресенье)

// Форматирование
now.toLocaleDateString('ru-RU');
now.toLocaleTimeString('ru-RU');

📝 Примеры из моей практики

Сумма цифр числа
function sumDigits(num) {
  let str = String(num);
  let sum = 0;
  for (let i = 0; i < str.length; i++) {
    sum += Number(str[i]);
  }
  return sum;
}

sumDigits(12345); // 15
Проверка на палиндром
function isPalindrome(str) {
  let reversed = str.split('').reverse().join('');
  return str === reversed;
}

isPalindrome('abcba'); // true
isPalindrome('hello'); // false
Удаление дублей из массива
function removeDuplicates(arr) {
  return arr.filter((item, index) => {
    return arr.indexOf(item) === index;
  });
}

removeDuplicates([1,2,2,3,3,4]);
// [1, 2, 3, 4]
